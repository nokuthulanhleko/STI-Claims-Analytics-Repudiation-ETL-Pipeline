-- DAX measures for STI Claims Dashboard
-- Paste these into Power BI (Modeling → New measure)

/* Basic counts and sums */

Total Claims = DISTINCTCOUNT('vw_claims_summary'[ClaimID])
Total Claim Amount = SUM('vw_claims_summary'[ClaimAmount])
Total Approved Amount = SUM('vw_claims_summary'[ApprovedAmount])
Average Claim Amount = AVERAGE('vw_claims_summary'[ClaimAmount])

/* Claims by status / coverage */
Claims By Status = COUNTROWS('vw_claims_summary')
Claims Approved = CALCULATE(COUNTROWS('vw_claims_summary'), 'vw_claims_summary'[ClaimStatus] = "Approved")
Claims Repudiated = CALCULATE(COUNTROWS('vw_claims_summary'), 'vw_claims_summary'[RepudiatedFlag] = 1)
Repudiation Rate = DIVIDE([Claims Repudiated], [Total Claims], 0)

/* Repudiation detail counts (from vw_repudiation_analysis) */
Forced Entry Repudiation Count = CALCULATE(COUNTROWS('vw_repudiation_analysis'), 'vw_repudiation_analysis'[ForcedEntryRepudiationFlag] = 1)
Proof Of Ownership Issue Count = CALCULATE(COUNTROWS('vw_repudiation_analysis'), 'vw_repudiation_analysis'[ProofOfOwnershipIssueFlag] = 1)

/* Fraud */
-- If FraudScore is 0-100 adjust threshold to 50; if 0-1 use 0.5
High Fraud Count = CALCULATE(COUNTROWS('vw_claims_summary'), 'vw_claims_summary'[FraudScore] >= 50)
High Fraud Rate = DIVIDE([High Fraud Count], [Total Claims], 0)
Average Fraud Score = AVERAGE('vw_claims_summary'[FraudScore])

/* Payouts and amounts */
Average Payout Ratio = AVERAGE('vw_claims_summary'[PayoutRatio])
Total Additional Excess = SUM('vw_claims_summary'[AdditionalExcessAmount])
Average Approved Amount = AVERAGE('vw_claims_summary'[ApprovedAmount])

/* Time intelligence */
Claims This Year = CALCULATE([Total Claims], FILTER('vw_claims_summary', 'vw_claims_summary'[YearNumber] = YEAR(TODAY())))
Claims Last Year = CALCULATE([Total Claims], FILTER('vw_claims_summary', 'vw_claims_summary'[YearNumber] = YEAR(TODAY())-1))
Claims YTD = TOTALYTD([Total Claim Amount], 'vw_claims_summary'[FullDate])

/* Ratios and KPIs */
Paid Ratio = DIVIDE([Total Approved Amount], [Total Claim Amount], 0)
Avg Claim per Policy = DIVIDE([Total Claim Amount], DISTINCTCOUNT('vw_claims_summary'[PolicyID]), 0)
